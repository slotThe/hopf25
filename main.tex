\begin{frame}\frametitle{The setup}
  Let \(\cat{C}\) be a monoidal,
  and \(\cat{M}, \cat{N}\) (left) \(\cat{C}\)-module categories.
  \uncover<2->{
    \begin{align*}
      \otimes \from \cat{C} \times \cat{C} \to \cat{C},
      \qquad\qquad
      \lact \from \cat{C} \times \cat{M} \to \cat{M}.
    \end{align*}
    \uncover<3->{
      such that for all \(x, y, z \in \cat{C}\) and \(m \in \cat{M}\), \eg,
      \[
        (x \otimes y) \otimes z \cong x \otimes (y \otimes z)
        \qquad\text{and}\qquad
        (x \otimes y) \lact m \cong x \lact (y \lact m).
      \]%
    }%
  }%
\end{frame}

\begin{frame}
  \begin{theorem}[Ostrik, Etingof--Gelaki--Nikshych--Ostrik]\label{finiterecon}
    \begin{itemize}[noitemsep]
      \item<1-> \(\cat{C}\) \alert<5>{finite} \alert<6>{tensor category};
      \item<2-> \(\cat{M}\) \alert<5>{finite} abelian \(\cat{C}\)\hyp{}module category; and
      \item<3-> the functor \(\blank \triangleright m \from \cat{C} \to \cat{M}\) is \alert<6>{exact} for all \(m \in \cat{M}\).
    \end{itemize}
    \uncover<4->{%
      Then there exists an algebra object \(A \in \cat{C}\)
      such that there is an equivalence of \(\cat{C}\)\hyp{}module categories \(\mathop{\mathrm{mod}}_{\cat{C}}A \simeq \cat{M}\).}
  \end{theorem}
  \begin{proposition*}<7>[Douglas--Schommer-Pries--Snyder]
    In the absence of rigidity,
    there are finite abelian \(\cat{C}\)-module categories that cannot be realised as the modules of an algebra object in \(\cat{C}\).
  \end{proposition*}
\end{frame}

\begin{frame}\frametitle{The Yoneda lemma™}
  \begin{proposition*}
    There is an equivalence of \(\cat{C}\)-module categories
    \begin{equation*}
      \cat{M} \simeq \mathsf{Str}\cat{C}\mathsf{Mod}(\cat{C},\cat{M}),\qquad
      \pause
      m \mapsto \alert<5>{\blank \lact m},\qquad
      \pause
      F1 \mapsfrom F.
    \end{equation*}
    \pause%
    In particular, \(\cat{C}^{\textcolor{gray}{\tensorop}} \simeq \mathsf{Str}\cat{C}\mathsf{Mod}(\cat{C},\cat{C})\).
  \end{proposition*}
  \pause%
\end{frame}

\begin{frame}[t]\frametitle{To Kelly and Beck}
  \vspace{.5cm}
  \begin{theorem}[\alt<-2>{Kelly}{Kelly, Halbig--Z}]
    Given an adjunction
    \alt<-2>{\(\adj{F}{U}{\cat{C}}{\cat{D}}\)}{\(\adj{F}{U}{\cat{M}}{\cat{N}}\)}
    between
    \alt<-2>{\alert<2>{monoidal} categories}{\(\cat{C}\)-\alert<3>{module} categories},
    oplax
    \alt<-2>{\alert<2>{monoidal}}{\(\cat{C}\)-\alert<3>{module}}
    structures on \(F\) are in bijective correspondence with
    lax \alt<-2>{\alert<2>{monoidal}}{\(\cat{C}\)-\alert<3>{module}}
    structures on \(U\).
  \end{theorem}
  \vspace{1cm}
  \begin{theorem}<5->[\alt<6>{Abelian monadicity}{Beck's monadicity theorem}]
    An adjunction \(\adj{F}{U}{\cat{C}}{\cat{D}}\) is monadic if\only<-5>{ and only if}
    \alt<6>{%
      \(U\)\! is exact and reflects zero objects.%
    }{%
      \(U\)\! is conservative, \(\cat{D}\) has coequalisers of \(U\)-split pairs, and \(U\) preserves them.%
    }%
  \end{theorem}
\end{frame}

\begin{frame}\frametitle{Internal projectives}
  Let \(\cat{C}\) and \(\cat{M}\) be abelian.
  \pause%
  An object \(\ell \in \cat{M}\) is \emph{closed} if there is an adjunction
  \[
    \adj{\blank \lact \ell}{\hom{\ell, \blank}}{\cat{C}}{\cat{M}}.
  \]
  \popup*<5>{(4.85,1.15)}{\(\adj{\blank \otimes x}{\blank \otimes \ld{\!x}}{\cat{C}}{\cat{C}}\)}%
  \pause%
  A closed object is called \emph{\(\cat{C}\)-projective} if \(\hom{\ell, \blank}\) is (right) exact\alt<-3>{.}{}
  \pause%
  and a \emph{\(\cat{C}\)-generator} if \(\hom{\ell, \blank}\) is faithful.
  \pause%
  \begin{example}
    \begin{itemize}
      \item Every object in a rigid monoidal category \(\cat{C}\) is \(\cat{C}\)-projective.\pause%
      \item Finite \(\cat{C}\)-module categories over finite tensor categories always\\
      admit \(\cat{C}\)-projective \(\cat{C}\)-generators.
    \end{itemize}
  \end{example}
\end{frame}

\begin{frame}\frametitle{A module structure for the Eilenberg–Moore category}
  \begin{theorem}[Linton, Day, Aguiar--Haim--López Franco, Stroiński--Z]
    The Eilenberg–Moore category of any right exact lax \(\cat{C}\)-module monad
    can be equipped with a canonical \(\cat{C}\)-module structure by means of Linton coequalisers.
  \end{theorem}
  \pause%
  \begin{definition}
    The \emph{Linton coequaliser} of \(x \in \cat{C}\) and \((m, \nabla_m) \in \cat{M}^T\) is:
    \[
      \begin{tikzcd}[ampersand replacement=\&]
        {T(x \lact Tm)} \& {T(x \lact m)} \& {x \blact m.}
        \arrow["\raisebox{0.3em}{\(T(x \,\lact\, \nabla_{m})\)}", shift left=1, from=1-1, to=1-2]
        \arrow["\raisebox{-0.3em}{\(\mu_{x \lact m}\circ TT_{2;x,m}\)}"', shift right=1, from=1-1, to=1-2]
        \arrow[two heads, from=1-2, to=1-3]
      \end{tikzcd}%
    \]
  \end{definition}
\end{frame}

\begin{frame}\frametitle{The reconstruction result}
  \begin{theorem}[Stroiński--Z]
    Let \(\cat{C}\) be an abelian monoidal category,
    \(\cat{M}\) an abelian \(\cat{C}\)\hyp{}module category,
    and assume that \(\ell \in \cat{M}\) is a closed \(\cat{C}\)\hyp{}projective \(\cat{C}\)\hyp{}generator.
    \pause%
    Then
    \vspace{-0.2cm}\[
      \cat{M} \simeq_{\lact} \cat{C}^{\hom{\ell, \blank \,\lact\, \ell}}.
    \]\vspace{-0.2cm}
    \pause%
    Furthermore, there is a bijection
    \vspace{-0.2cm}\begin{align*}
      \faktor{\{(\cat{M},\ell)\ \text{as above}\}}{\cat{M} \simeq \cat{N}}
      &\xleftrightarrow{\ \cong\ \,}
        \Bigg\{
        \begin{aligned}
          &\text{Right exact lax \(\cat{C}\)\hyp{}module}\\
          &\text{monads on \(\cat{C}\)}
        \end{aligned}
        \Bigg\}
        /\, \cat{C}^T \!\simeq\! \cat{C}^S
    \end{align*}
  \end{theorem}
\end{frame}

\begin{frame}\frametitle{Deducing a theorem for Hopf trimodules}
  \begin{proposition*}
    Let \(\cat{C}\) be a left closed monoidal category such that the canonical embedding
    \[
      \mathsf{Str}\cat{C}\mathsf{Mod}(\cat{C},\cat{C}) \longhookrightarrow \mathsf{Lax}\cat{C}\mathsf{Mod}(\cat{C},\cat{C})
    \]
    is an equivalence.
    \pause%
    Then \(\cat{C}\) is left rigid.
  \end{proposition*}
  \pause%
  \begin{corollary}[Stroiński--Z]
    A bialgebra \(B\) admits a twisted antipode if and only if the canonical functor
    \(B \kotimes \blank\from \Tetramod[B] \to \Tetramod[B][B][][B]\) is an equivalence.
  \end{corollary}
\end{frame}

\begin{frame}\frametitle{Fusion operators for Hopf monads}
  \begin{proposition*}[Stroiński--Z]
    The bimonad \(T \defeq UF\) of an oplax monoidal adjunction \(\adj{F}{U}{\cat{C}}{\cat{D}}\)
    is canonically an oplax \(\cat{D}\)\hyp{}module monad.

    \pause%
    The right fusion operator is the ``free part'' of the coherence morphism:
    \[
      T_{2;F,\Id} = T_{\mathsf{rf}} \from T(T \otimes \Id) \nt T \otimes T \Alt<2>{.}{,}
    \]
    \pause%
    and \(T_{\mathsf{rf}}\) is an isomorphism if and only if \(\,T_2\) is.
  \end{proposition*}
\end{frame}
